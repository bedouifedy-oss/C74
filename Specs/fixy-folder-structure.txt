# ============================================
# INSTALLATION COMMANDS
# ============================================

# Initialize Next.js with TypeScript
npx create-next-app@latest c74-tn --typescript --tailwind --app --src-dir

# Install ShadCN UI
npx shadcn-ui@latest init

# Install i18n
npm install next-intl

# Install RTL support
npm install tailwindcss-rtl -D

# Install form handling
npm install react-hook-form @hookform/resolvers zod

# Install Supabase
npm install @supabase/supabase-js

# Install state management
npm install zustand

# Install utilities
npm install clsx tailwind-merge
npm install date-fns
npm install lucide-react

# Install dev dependencies
npm install -D @types/node

# ============================================
# NEXT STEPS
# ============================================

1. Run: npx shadcn-ui@latest init
2. Add base components:
   npx shadcn-ui@latest add button card input label badge avatar
3. Create folder structure (copy from above)
4. Copy Tailwind config
5. Start building components
